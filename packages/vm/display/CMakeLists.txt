cmake_minimum_required(VERSION 3.22)
project(DockpipeLocalDisplay VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Windows|Darwin)$")
    message(FATAL_ERROR "Display adapters exist only for Linux, Windows and macOS")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10.3 EXACT REQUIRED COMPONENTS Widgets Test)

if(WIN32)
    set(DOCKPIPE_GIO_PREFIX "" CACHE PATH "Explicit task-local Windows GLib/GIO prefix")
    if(NOT IS_ABSOLUTE "${DOCKPIPE_GIO_PREFIX}")
        message(FATAL_ERROR "Windows requires an explicit DOCKPIPE_GIO_PREFIX")
    endif()
    find_path(GIO_INCLUDE gio/gio.h PATHS "${DOCKPIPE_GIO_PREFIX}/include/glib-2.0" NO_DEFAULT_PATH REQUIRED)
    find_path(GIO_CONFIG_INCLUDE glibconfig.h PATHS "${DOCKPIPE_GIO_PREFIX}/lib/glib-2.0/include" NO_DEFAULT_PATH REQUIRED)
    add_library(display-gio INTERFACE)
    target_include_directories(display-gio INTERFACE "${GIO_INCLUDE}" "${GIO_CONFIG_INCLUDE}")
    foreach(name IN ITEMS gio gobject glib)
        find_library(GIO_${name}_LIBRARY NAMES "${name}-2.0" PATHS "${DOCKPIPE_GIO_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
        target_link_libraries(display-gio INTERFACE "${GIO_${name}_LIBRARY}")
    endforeach()
    set(platform_sources src/NativeDisplayPlatform_windows.cpp src/QemuListenerPeer_windows.cpp src/SharedFrameSource_windows.cpp)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0>=2.66)
    add_library(display-gio INTERFACE)
    target_link_libraries(display-gio INTERFACE PkgConfig::GIO)
    set(platform_sources src/NativeDisplayPlatform_unix.cpp src/QemuListenerPeer_unix.cpp src/SharedFrameSource.cpp)
endif()
add_library(dockpipe-display-widget STATIC src/DisplayWidget.cpp src/FrameStore.cpp src/NativeKeyMap.cpp)
target_include_directories(dockpipe-display-widget PUBLIC src)
target_link_libraries(dockpipe-display-widget PUBLIC Qt6::Widgets)
add_library(dockpipe-qemu-dbus STATIC src/QemuDisplayTransport.cpp src/QmpAbsoluteInput.cpp src/DisplayPointerStateStore.cpp src/DisplayEndpoint.cpp src/DisplayAddress.cpp src/RawFrameAssembler.cpp ${platform_sources})
target_link_libraries(dockpipe-qemu-dbus PUBLIC dockpipe-display-widget PRIVATE display-gio)
if(WIN32)
    target_compile_definitions(dockpipe-qemu-dbus PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(dockpipe-qemu-dbus PRIVATE ws2_32 advapi32)
elseif(APPLE)
    target_link_libraries(dockpipe-qemu-dbus PRIVATE proc)
endif()
add_executable(dockpipe-local-display src/main.cpp)
target_link_libraries(dockpipe-local-display PRIVATE dockpipe-qemu-dbus)

enable_testing()
function(display_test name source)
    add_executable(${name}-test ${source})
    target_link_libraries(${name}-test PRIVATE dockpipe-qemu-dbus Qt6::Test)
    add_test(NAME ${name} COMMAND ${name}-test)
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 10)
endfunction()
display_test(qmp-absolute-input tests/QmpAbsoluteInputTest.cpp)
target_link_libraries(qmp-absolute-input-test PRIVATE display-gio)
display_test(frame-store tests/FrameStoreTest.cpp)
display_test(native-key-map tests/NativeKeyMapTest.cpp)
display_test(display-address tests/DisplayAddressTest.cpp)
display_test(raw-frame-assembler tests/RawFrameAssemblerTest.cpp)
display_test(transport-lifecycle tests/TransportLifecycleTest.cpp)
display_test(display-widget-input tests/DisplayWidgetInputTest.cpp)
display_test(display-pointer-state tests/DisplayPointerStateTest.cpp)
if(WIN32)
    display_test(native-windows tests/WindowsPlatformTest.cpp)
    target_link_libraries(native-windows-test PRIVATE advapi32)
else()
    display_test(shared-frame-source tests/SharedFrameSourceTest.cpp)
    display_test(display-endpoint tests/DisplayEndpointTest.cpp)
endif()

option(DOCKPIPE_BUILD_DISPLAY_PLUGIN "Build the installed host SDK adapter" OFF)
set(DOCKPIPE_DISPLAY_SDK "launcher" CACHE STRING "Explicit installed SDK contract: launcher or display")
set_property(CACHE DOCKPIPE_DISPLAY_SDK PROPERTY STRINGS launcher display)
if(DOCKPIPE_BUILD_DISPLAY_PLUGIN)
    if(DOCKPIPE_DISPLAY_SDK STREQUAL "launcher")
        find_package(DockPipeLauncherSDK 1.0.0 EXACT REQUIRED)
        set(display_headers DockPipe::LauncherHeaders)
    elseif(DOCKPIPE_DISPLAY_SDK STREQUAL "display")
        find_package(DockPipeDisplaySDK 1.0.0 EXACT REQUIRED)
        set(display_headers DockPipe::DisplayHeaders)
    else()
        message(FATAL_ERROR "DOCKPIPE_DISPLAY_SDK must explicitly select launcher or display")
    endif()
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
        set(display_cpu "arm64")
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
        set(display_cpu "x86_64")
    else()
        message(FATAL_ERROR "Display metadata currently supports only amd64 and arm64")
    endif()
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Display adapters currently require a 64-bit host")
    endif()
    if(WIN32)
        set(display_build_abi "${display_cpu}-little_endian-llp64")
    else()
        set(display_build_abi "${display_cpu}-little_endian-lp64")
    endif()
    configure_file(src/display-metadata.json.in display-metadata.json @ONLY)
    set_target_properties(dockpipe-display-widget dockpipe-qemu-dbus PROPERTIES POSITION_INDEPENDENT_CODE ON)
    add_library(dockpipe-vm-display MODULE src/DisplayPackage.cpp)
    target_include_directories(dockpipe-vm-display PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(dockpipe-vm-display PRIVATE ${display_headers} dockpipe-qemu-dbus)
    set_target_properties(dockpipe-vm-display PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/package/assets/display"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/package/assets/display"
        RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/package/assets/display"
        LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/package/assets/display")
    option(DOCKPIPE_BUILD_FACTORY_PROOF "Build the minimal installed-interface qualification harness" OFF)
    if(DOCKPIPE_BUILD_FACTORY_PROOF)
        add_executable(display-factory-proof tooling/FactoryProof.cpp)
        target_link_libraries(display-factory-proof PRIVATE ${display_headers} Qt6::Widgets)
        target_compile_definitions(display-factory-proof PRIVATE
            DISPLAY_PROOF_COMPILER_ID="${CMAKE_CXX_COMPILER_ID}"
            DISPLAY_PROOF_COMPILER_VERSION="${CMAKE_CXX_COMPILER_VERSION}")
    endif()
    if(WIN32)
        set_target_properties(dockpipe-vm-display PROPERTIES PREFIX "")
    elseif(APPLE)
        set_target_properties(dockpipe-vm-display PROPERTIES SUFFIX ".dylib"
            BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@loader_path/lib")
    else()
        set_target_properties(dockpipe-vm-display PROPERTIES
            BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
        target_link_options(dockpipe-vm-display PRIVATE "-Wl,--disable-new-dtags")
    endif()
endif()
