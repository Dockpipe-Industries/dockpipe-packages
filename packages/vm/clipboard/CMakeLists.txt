cmake_minimum_required(VERSION 3.22)
project(DockpipeVmClipboard LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
find_package(Qt6 6.10.3 EXACT REQUIRED COMPONENTS Core Gui Network)
add_executable(dockpipe-vm-clipboard src/main.cpp src/ClipboardBridge.cpp src/ClipboardProof.cpp)
target_link_libraries(dockpipe-vm-clipboard PRIVATE Qt6::Core Qt6::Gui Qt6::Network)
install(TARGETS dockpipe-vm-clipboard RUNTIME DESTINATION bin)
